import os

class DatabaseSetup:
    def __init__(self, db_path="database/auth_system.db"):
        """Initialize database connection"""
        # Create database directory if it doesn't exist
        os.makedirs(os.path.dirname(db_path), exist_ok=True)
        
        self.db_path = db_path
        self.conn = None
        self.cursor = None
    
    def connect(self):
        """Establish database connection"""
        self.conn = sqlite3.connect(self.db_path)
        self.conn.row_factory = sqlite3.Row  # This enables column access by name
        self.cursor = self.conn.cursor()
        print(f"✅ Connected to database: {self.db_path}")
    
    def close(self):
        """Close database connection"""
        if self.conn:
            self.conn.commit()
            self.conn.close()
            print("✅ Database connection closed")
    
    def enable_foreign_keys(self):
        """Enable foreign key constraints in SQLite"""
        self.cursor.execute("PRAGMA foreign_keys = ON;")
        print("✅ Foreign key constraints enabled")
    
    def create_tables(self):

"""
Database Utility Module - Use this for all database operations in your app
"""

import sqlite3
import json
from contextlib import contextmanager
from datetime import datetime
import os

class Database:
    def __init__(self, db_path="database/auth_system.db"):
        self.db_path = db_path
        
        # Ensure database directory exists
        os.makedirs(os.path.dirname(db_path), exist_ok=True)
    
    @contextmanager
    def get_connection(self):
        """Context manager for database connections"""
        conn = sqlite3.connect(self.db_path)
        conn.row_factory = sqlite3.Row
        conn.execute("PRAGMA foreign_keys = ON")
        try:
            yield conn
        finally:
            conn.close()
    
    def execute_query(self, query, params=(), fetch_one=False, fetch_all=False):
        """Execute a query and return results"""
        with self.get_connection() as conn:
            cursor = conn.cursor()
            cursor.execute(query, params)
            conn.commit()
            
            if fetch_one:
                return cursor.fetchone()
            elif fetch_all:
                return cursor.fetchall()
            else:
                return cursor.lastrowid
    
    # ============= Company Operations =============
    
    def register_company(self, company_data):
        """Register a new company (PENDING status)"""
        query = '''
            INSERT INTO companies 
            (company_name, license_number, registration_number, contact_email, 
             contact_phone, address, website, public_key, public_key_parameters)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
        '''
        return self.execute_query(query, (
            company_data['company_name'],
            company_data['license_number'],
            company_data.get('registration_number'),
            company_data['contact_email'],
            company_data.get('contact_phone'),
            company_data.get('address'),
            company_data.get('website'),
            company_data['public_key'],
            company_data.get('public_key_parameters')
        ))
    
    def approve_company(self, company_id, admin_id):
        """Approve a company after verification"""
        query = '''
            UPDATE companies 
            SET status = 'APPROVED', 
                approved_by = ?,
                approved_at = CURRENT_TIMESTAMP,
                updated_at = CURRENT_TIMESTAMP
            WHERE company_id = ? AND status = 'PENDING'
        '''
        return self.execute_query(query, (admin_id, company_id))
    
    def revoke_company(self, company_id, admin_id, reason):
        """Revoke a company's trust"""
        query = '''
            UPDATE companies 
            SET status = 'REVOKED',
                revoked_by = ?,
                revoked_at = CURRENT_TIMESTAMP,
                revocation_reason = ?,
                updated_at = CURRENT_TIMESTAMP
            WHERE company_id = ?
        '''
        return self.execute_query(query, (admin_id, reason, company_id))
    
    def get_approved_companies(self):
        """Get all approved companies (for gate sync)"""
        query = '''
            SELECT company_id, company_name, public_key, public_key_parameters,
                   updated_at
            FROM companies
            WHERE status = 'APPROVED'
        '''
        with self.get_connection() as conn:
            cursor = conn.cursor()
            cursor.execute(query)
            return [dict(row) for row in cursor.fetchall()]
    
    # ============= Worker Operations =============
    
    def register_worker(self, worker_data):
        """Register a new worker"""
        query = '''
            INSERT INTO workers 
            (worker_identifier, full_name, email, phone, company_id, 
             employee_id, role, attributes_json, expires_at)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)

        return self.execute_query(query, (
            worker_data['worker_identifier'],

            worker_data.get('email'),
            worker_data.get('phone'),
            worker_data['company_id'],
            worker_data.get('employee_id'),
            worker_data.get('role', 'delivery'),
            json.dumps(worker_data.get('attributes', {})),
            worker_data.get('expires_at')
        ))
    
    def issue_credential(self, worker_id, mu_value, sa_values):
        """Issue a credential to a worker (from paper's CreGen)"""
        
        # Calculate credential hash for quick lookup
        import hashlib
        cred_string = f"{worker_id}:{mu_value}:{sa_values}"
        cred_hash = hashlib.sha256(cred_string.encode()).hexdigest()
        
        query = '''
            INSERT INTO credentials 
            (worker_id, mu_value, sa_values, credential_hash, expires_at)
            VALUES (?, ?, ?, ?, DATE('now', '+30 days'))
        '''
        return self.execute_query(query, (
            worker_id, mu_value, json.dumps(sa_values), cred_hash
        ))
    
    def revoke_worker(self, worker_id, admin_id, reason):
        """Revoke a worker's access"""
        query = '''
            UPDATE workers 
            SET is_revoked = 1,
                revoked_at = CURRENT_TIMESTAMP,
                revoked_by = ?,
                revocation_reason = ?
            WHERE worker_id = ?
        '''
        return self.execute_query(query, (admin_id, reason, worker_id))
    
    # ============= Pseudonym Operations =============
    
    def register_pseudonym(self, credential_id, pu_value, pa_values, z_value):
        """Register a generated pseudonym (from PseuGen)"""
        
        # Create hash for lookup
        import hashlib
        pseudonym_hash = hashlib.sha256(f"{pu_value}:{z_value}".encode()).hexdigest()
        
        query = '''
            INSERT INTO pseudonyms 
            (credential_id, pu_value, pa_values, z_value, pseudonym_hash, expires_at)
            VALUES (?, ?, ?, ?, ?, DATETIME('now', '+5 minutes'))
        '''
        return self.execute_query(query, (
            credential_id, pu_value, json.dumps(pa_values), z_value, pseudonym_hash
        ))
    
    def check_pseudonym_replay(self, pu_value, z_value):
        """Check if pseudonym has been used before (replay prevention)"""
        query = '''
            SELECT is_consumed, use_count 
            FROM pseudonyms 
            WHERE pu_value = ? AND z_value = ?
        '''
        result = self.execute_query(query, (pu_value, z_value), fetch_one=True)
        return result is not None and result['is_consumed'] == 1

from sqlalchemy.ext.asyncio import (
    AsyncSession,

    async_sessionmaker,
    AsyncEngine
)
from sqlalchemy.orm import sessionmaker

from app.config import settings

logger = logging.getLogger(__name__)

class DatabaseManager:
    """Manages database connections and sessions"""
    
    def __init__(self):
        self.engine: Optional[AsyncEngine] = None
        self.async_session_maker: Optional[async_sessionmaker[AsyncSession]] = None
        self._initialized = False
    
    async def initialize(self):
        """Initialize database engine and session maker"""
        if self._initialized:
            return
        
        # Configure engine based on database type
        if settings.DATABASE_URL.startswith("sqlite"):
            # SQLite specific configuration
            self.engine = create_async_engine(
                settings.DATABASE_URL,
                echo=settings.LOG_LEVEL == "DEBUG",
                future=True,
                poolclass=NullPool,  # SQLite doesn't need connection pooling
                connect_args={
                    "check_same_thread": False,  # Allow multi-thread access
                    "timeout": 30,  # 30 second timeout
                }
            )
        else:
            # PostgreSQL/MySQL configuration
            self.engine = create_async_engine(
                settings.DATABASE_URL,
                echo=settings.LOG_LEVEL == "DEBUG",
                future=True,
                pool_size=settings.DATABASE_POOL_SIZE,
                max_overflow=10,
                pool_pre_ping=True,  # Verify connections before using
                pool_recycle=3600,  # Recycle connections after 1 hour
            )
        
        self.async_session_maker = async_sessionmaker(
            self.engine,
            class_=AsyncSession,
            expire_on_commit=False,
            autocommit=False,
            autoflush=False,
        )
        
        self._initialized = True
        logger.info(f"Database initialized: {settings.DATABASE_URL}")
    
    async def close(self):
        """Close database connections"""
        if self.engine:
            await self.engine.dispose()
            self._initialized = False
            logger.info("Database connections closed")
    
    @contextlib.asynccontextmanager
    async def get_session(self) -> AsyncGenerator[AsyncSession, None]:
        """Get a database session (context manager)"""
        if not self._initialized:
            await self.initialize()
        
        async with self.async_session_maker() as session:
            try:
                yield session
                await session.commit()
            except Exception as e:
                await session.rollback()
                logger.error(f"Database session error: {e}")
                raise
            finally:
                await session.close()
    
    async def get_session_direct(self) -> AsyncSession:
        """Get a database session (direct, must be closed manually)"""
        if not self._initialized:
            await self.initialize()
        
        session = self.async_session_maker()
        return session

        # Get generators
        self.P = self.group.random(G1)  # Public generator P
        self.g = pair(self.P, self.P)  # g = e(P, P)
        self.h = self.group.random(GT)  # Another generator for GT
        self.order = self.group.order()

@dataclass
class PublicParameters:

    security_param: int
    group_order: int

    generator_g: str  # Serialized g

        """Deserialize from JSON"""
        return cls(**json.loads(data))

@dataclass
class AttributeKeyPair:
    """Key pair for a single attribute"""
    attribute_name: str
    public_key: str  # W_i = s_i * P
    secret_key: str  # s_i (master secret)
    
    def to_dict(self) -> Dict:
        return {
            "attr": self.attribute_name,
            "pk": self.public_key,
            "sk": self.secret_key
        }

class KeyGen:
    """
    Key Generation Algorithm (KeyGen) from paper Section 5.
    
    Input: security parameter κ
    Output: 
        - Public parameters PP = {G1, GT, Zp, P, g, h, e(), H()}
        - Public keys per attribute {W1, ..., Wn}
        - Master secret keys per attribute {s1, ..., sn}
    """
    
    def __init__(self, pairing: BilinearPairing = None):
        self.pairing = pairing or get_pairing()
        self.P, self.g, self.h = self.pairing.get_generators()
    
    def generate(self, attributes: List[str]) -> Tuple[PublicParameters, Dict[str, AttributeKeyPair]]:
        """
        Execute KeyGen algorithm.
        
        Steps from paper:
        1) selects a public generator P ← R G1
        2) selects public generators h and g ← R GT
        3) selects a secret value per attribute s_i ← R Zp ∀ i ∈ {1,...,n}
        4) computes a public key per attribute W_i = s_i P ∀ i ∈ {1,...,n}
        """
        logger.info(f"Generating keys for {len(attributes)} attributes")
        
        # Create public parameters
        pp = PublicParameters(
            curve_type=settings.CURVE_TYPE,
            security_param=settings.SECURITY_PARAMETER,
            group_order=settings.GROUP_ORDER,
            generator_P=self.pairing.serialize_G1(self.P),
            generator_g=self.pairing.serialize_GT(self.g),
            generator_h=self.pairing.serialize_GT(self.h)
        )
        
        # Generate key pairs for each attribute
        attribute_keys = {}
        
        for attr in attributes:
            # Step 3: select secret s_i
            s_i = self.pairing.random_ZR()
            
            # Step 4: compute W_i = s_i * P
            W_i = self._scalar_mult_G1(s_i, self.P)
            
            attribute_keys[attr] = AttributeKeyPair(
                attribute_name=attr,
                public_key=self.pairing.serialize_G1(W_i),
                secret_key=self._serialize_ZR(s_i)
            )
            
            logger.debug(f"Generated keys for attribute: {attr}")
        
        logger.info(f"KeyGen completed for {len(attributes)} attributes")
        return pp, attribute_keys
    
    def _scalar_mult_G1(self, scalar: Any, point: Any) -> Any:
        """Multiply a G1 point by a scalar"""
        if hasattr(scalar, '__mul__'):
            return scalar * point
        else:
            # For mock implementation
            return point * scalar
    
    def _serialize_ZR(self, element: Any) -> str:
        """Serialize ZR element to string"""
        if hasattr(element, '__str__'):
            return str(element)
        return str(element)
    
    def generate_for_company(self, company_name: str, attributes: List[str]) -> Dict:
        """
        Generate keys for a company and return in format for database storage.
        """
        pp, attr_keys = self.generate(attributes)
        
        # Format for CompanyPublicKey table
        result = {
            "public_key": pp.to_json(),
            "attribute_keys": {
                attr: {
                    "public_key": keys.public_key,
                    "secret_key": keys.secret_key
                }
                for attr, keys in attr_keys.items()
            },
            "key_version": 1
        }
        
        return result

        - Set S of k attributes S = {att_i1, ..., att_ik}
    
    Output:
        - Credential cred = (μ, Sa) where Sa = {Sa_i1, ..., Sa_ik}
          and Sa_i = P * 1/(s_i + μ)
    """
    
    def __init__(self, pairing: BilinearPairing = None):
        self.pairing = pairing or get_pairing()
        self.P, self.g, self.h = self.pairing.get_generators()
    
    def generate(
        self, 
        attribute_secrets: Dict[str, Any],  # Map attribute -> s_i
        worker_id: int,
        company_id: int,
        attributes: List[str],
        expiry_days: int = 30
    ) -> Credential:
        """
        Generate a credential for a worker.
        
        Steps from paper:
        1) randomly selects a secret value μ ← R Zp
        2) computes secret Sa_i = P * 1/(s_i + μ) for all i in S
        3) credential is the tuple cred = (μ, Sa)
        """
        logger.info(f"Generating credential for worker {worker_id} with {len(attributes)} attributes")
        
        # Step 1: select random μ
        mu = self.pairing.random_ZR()
        mu_str = self._serialize_ZR(mu)
        
        # Step 2: compute Sa_i for each attribute
        sa_values = {}
        
        for attr in attributes:
            if attr not in attribute_secrets:
                raise ValueError(f"Attribute {attr} not found in company's secrets")
            
            # Get s_i for this attribute
            s_i = self._deserialize_ZR(attribute_secrets[attr])
            
            # Compute s_i + μ in Zp
            s_i_plus_mu = self._add_zr(s_i, mu)
            
            # Compute 1/(s_i + μ)
            inv_s_i_plus_mu = self._inverse_zr(s_i_plus_mu)
            
            # Compute Sa_i = inv(s_i + μ) * P
            Sa_i = self._scalar_mult_G1(inv_s_i_plus_mu, self.P)
            
            sa_values[attr] = self.pairing.serialize_G1(Sa_i)
            
            logger.debug(f"Generated Sa for attribute {attr}")
        
        # Step 3: create credential
        from datetime import datetime, timedelta
        issued_at = datetime.utcnow().isoformat()
        expires_at = (datetime.utcnow() + timedelta(days=expiry_days)).isoformat()
        
        credential = Credential(
            mu=mu_str,
            sa_values=sa_values,
            worker_id=worker_id,
            company_id=company_id,
            issued_at=issued_at,
            expires_at=expires_at
        )
        
        logger.info(f"Credential generated with hash: {credential.get_hash()}")
        return credential
    
    def verify_credential(self, credential: Credential, public_keys: Dict[str, str]) -> bool:
        """
        Verify credential correctness by checking:
        e(μP + W_i, Sa_i) = e(P, P) for all i
        """
        mu = self._deserialize_ZR(credential.mu)
        
        # Compute μP
        mu_P = self._scalar_mult_G1(mu, self.P)
        
        for attr, sa_str in credential.sa_values.items():
            if attr not in public_keys:
                logger.warning(f"Attribute {attr} not found in public keys")
                return False
            
            # Get W_i (public key)
            W_i = self.pairing.deserialize_G1(public_keys[attr])
            
            # Get Sa_i
            Sa_i = self.pairing.deserialize_G1(sa_str)
            
            # Compute μP + W_i
            mu_P_plus_W_i = self._add_g1(mu_P, W_i)
            
            # Check e(μP + W_i, Sa_i) == e(P, P)
            left = self.pairing.pair(mu_P_plus_W_i, Sa_i)
            right = self.pairing.pair(self.P, self.P)
            
            # Compare (need proper equality check)
            if not self._equal_gt(left, right):
                logger.warning(f"Credential verification failed for attribute {attr}")
                return False
        
        logger.info("Credential verification successful")
        return True
    
    def _scalar_mult_G1(self, scalar: Any, point: Any) -> Any:
        """Multiply G1 point by scalar"""
        if hasattr(scalar, '__mul__'):
            return scalar * point
        return point * scalar
    
    def _add_g1(self, a: Any, b: Any) -> Any:
        """Add two G1 points"""
        if hasattr(a, '__add__'):
            return a + b
        return a + b
    
    def _add_zr(self, a: Any, b: Any) -> Any:
        """Add two ZR elements"""
        if hasattr(a, '__add__'):
            return a + b
        return a + b
    
    def _inverse_zr(self, a: Any) -> Any:
        """Compute multiplicative inverse in Zp"""
        if hasattr(a, '__invert__'):
            return ~a
        return 1/a
    
    def _serialize_ZR(self, element: Any) -> str:
        """Serialize ZR element"""
        return str(element)
    
    def _deserialize_ZR(self, data: str) -> Any:
        """Deserialize ZR element"""
        # In real implementation, convert from string back to ZR
        return data
    
    def _equal_gt(self, a: Any, b: Any) -> bool:
        """Compare two GT elements"""
        if hasattr(a, '__eq__'):
            return a == b
        return str(a) == str(b)

class Pseudonym:
    """
    Pseudonym from paper Section 5: pseu_S' = (Pu, Pa1, ..., Pan, z)
    where:
        - Pu = (μ + μ')P
        - Pa_i = μ' * Sa_i for attributes in credential
        - Pa_j = (μ_j - (μ + μ')) * Sa_j for delegated attributes
        - Pa_i = a_i * P for simulated attributes
        - z = random public value
    """
    
    def __init__(
        self,
        pu_value: str,
        pa_values: Dict[str, str],
        z_value: str,
        mu_prime: str,  # Secret μ' (not part of public pseudonym)
        credential_id: int,
        expires_at: Optional[str] = None
    ):
        self.pu_value = pu_value
        self.pa_values = pa_values
        self.z_value = z_value
        self.mu_prime = mu_prime  # This is secret, not shared
        self.credential_id = credential_id
        self.expires_at = expires_at
    
    def to_dict(self) -> Dict:
        """Convert to dict for JSON serialization (public part only)"""
        return {
            "pu": self.pu_value,
            "pa": self.pa_values,
            "z": self.z_value,
            "expires_at": self.expires_at
        }
    
    def get_hash(self) -> str:
        """Generate pseudonym hash for lookup"""
        hash_input = f"{self.pu_value}:{self.z_value}"
        return hashlib.sha256(hash_input.encode()).hexdigest()

class PseuGen:
    """
    Pseudonym Generation Algorithm (PseuGen) from paper Section 5.
    
    Input:
        - Credential cred = (μ, Sa) for attribute set S
        - Optional delegated attributes (μ_j, Sa_j) for set S_d
        - Public parameters PP
    
    Output:
        - Pseudonym pseu_S' for attributes S' = S ∪ S_d
        - Secret value μ' associated with pseudonym
    """
    
    def __init__(self, pairing: BilinearPairing = None):
        self.pairing = pairing or get_pairing()
        self.P, self.g, self.h = self.pairing.get_generators()
    
    def generate(
        self,
        credential: Credential,
        all_attributes: List[str],  # All possible attributes in system
        delegated_attributes: Optional[Dict[str, Tuple[Any, Any]]] = None,  # attr -> (μ_j, Sa_j)
        z_value: Optional[str] = None,
        expiry_minutes: int = 5
    ) -> Tuple[Pseudonym, Any]:
        """
        Generate a pseudonym for a subset of attributes.
        
        Steps from paper:
        1) picks a random public value z ← R Zp
        2) computes d = H(z)
        3) computes secret value μ' = (d - μ) / 2
        4) computes Pu = (μ + μ')P
        5) computes Pa_i = μ' * Sa_i for i in I_S (own attributes)
        6) computes Pa_j = (μ_j - (μ + μ')) * Sa_j for j in I_Sd (delegated)
        7) computes random simulated Pa_i = a_i * P for rest of attributes
        """
        logger.info(f"Generating pseudonym for credential {credential.get_hash()}")
        
        # Step 1: pick random z (or use provided)
        if z_value is None:
            z_raw = secrets.token_bytes(32)
            z_value = self._serialize_ZR(self.pairing.hash_to_ZR(z_raw))
        
        z = self._deserialize_ZR(z_value)
        
        # Step 2: compute d = H(z)
        z_bytes = str(z_value).encode()
        d = self.pairing.hash_to_ZR(z_bytes)
        
        # Step 3: compute μ' = (d - μ) / 2
        mu = self._deserialize_ZR(credential.mu)
        d_minus_mu = self._subtract_zr(d, mu)
        mu_prime = self._divide_zr_by_2(d_minus_mu)
        mu_prime_str = self._serialize_ZR(mu_prime)
        
        # Step 4: compute Pu = (μ + μ')P
        mu_plus_mu_prime = self._add_zr(mu, mu_prime)
        Pu = self._scalar_mult_G1(mu_plus_mu_prime, self.P)
        pu_str = self.pairing.serialize_G1(Pu)
        
        # Step 5: compute Pa_i for own attributes
        pa_values = {}
        
        # Deserialize own Sa values
        own_sa = {}
        for attr, sa_str in credential.sa_values.items():
            own_sa[attr] = self.pairing.deserialize_G1(sa_str)
        
        # Step 6: prepare delegated attributes
        delegated = delegated_attributes or {}
        delegated_sa = {}
        for attr, (mu_j, sa_j) in delegated.items():
            delegated_sa[attr] = {
                'mu_j': mu_j,
                'sa_j': sa_j
            }
        
        # Compute Pa for each attribute in the universe
        for attr in all_attributes:
            if attr in own_sa:
                # Own attribute: Pa_i = μ' * Sa_i
                Sa_i = own_sa[attr]
                Pa_i = self._scalar_mult_G1(mu_prime, Sa_i)
                pa_values[attr] = self.pairing.serialize_G1(Pa_i)
                logger.debug(f"Computed Pa for own attribute: {attr}")
                
            elif attr in delegated_sa:
                # Delegated attribute: Pa_j = (μ_j - (μ + μ')) * Sa_j
                mu_j = delegated_sa[attr]['mu_j']
                Sa_j = delegated_sa[attr]['sa_j']
                
                # Compute μ_j - (μ + μ')
                mu_plus_mu_prime = self._add_zr(mu, mu_prime)
                mu_j_minus = self._subtract_zr(mu_j, mu_plus_mu_prime)
                
                # Compute Pa_j
                Pa_j = self._scalar_mult_G1(mu_j_minus, Sa_j)
                pa_values[attr] = self.pairing.serialize_G1(Pa_j)
                logger.debug(f"Computed Pa for delegated attribute: {attr}")
                
            else:
                # Simulated attribute: Pa_i = a_i * P (random)
                a_i = self.pairing.random_ZR()
                Pa_i = self._scalar_mult_G1(a_i, self.P)
                pa_values[attr] = self.pairing.serialize_G1(Pa_i)
                logger.debug(f"Computed simulated Pa for attribute: {attr}")
        
        # Set expiry
        expires_at = (datetime.utcnow() + timedelta(minutes=expiry_minutes)).isoformat()
        
        # Create pseudonym
        pseudonym = Pseudonym(
            pu_value=pu_str,
            pa_values=pa_values,
            z_value=z_value,
            mu_prime=mu_prime_str,
            credential_id=credential.credential_id if hasattr(credential, 'credential_id') else 0,
            expires_at=expires_at
        )
        
        logger.info(f"Pseudonym generated with hash: {pseudonym.get_hash()}")
        return pseudonym, mu_prime
    
    def _scalar_mult_G1(self, scalar: Any, point: Any) -> Any:
        """Multiply G1 point by scalar"""
        if hasattr(scalar, '__mul__'):
            return scalar * point
        return point * scalar
    
    def _add_zr(self, a: Any, b: Any) -> Any:
        """Add two ZR elements"""
        if hasattr(a, '__add__'):
            return a + b
        return a + b
    
    def _subtract_zr(self, a: Any, b: Any) -> Any:
        """Subtract two ZR elements"""
        if hasattr(a, '__sub__'):
            return a - b
        return a - b
    
    def _divide_zr_by_2(self, a: Any) -> Any:
        """Divide ZR element by 2"""
        inv_2 = self._inverse_zr(2)
        if hasattr(a, '__mul__'):
            return a * inv_2
        return a * inv_2
    
    def _inverse_zr(self, a: Any) -> Any:
        """Compute multiplicative inverse"""
        if hasattr(a, '__invert__'):
            return ~a
        return 1/a
    
    def _serialize_ZR(self, element: Any) -> str:
        """Serialize ZR element"""
        return str(element)
    
    def _deserialize_ZR(self, data: str) -> Any:
        """Deserialize ZR element"""
        return data

"""
Access Tree Structure implementation from Section 6 of the paper.
Implements monotone access structures and secret sharing for attribute-based policies.
"""

from typing import List, Dict, Any, Optional, Set, Tuple
from enum import Enum
import json
import logging
from dataclasses import dataclass, field

from app.crypto.secretsharing import ShamirSecretSharing

logger = logging.getLogger(__name__)

class GateType(Enum):
    """Types of threshold gates in access tree"""
    OR = "OR"  # k = 1
    AND = "AND"  # k = n
    THRESHOLD = "THRESHOLD"  # 1 < k < n

@dataclass
class AccessTreeNode:
    """Node in the access tree"""
    node_id: str
    gate_type: GateType
    threshold: Optional[int] = None  # k for threshold gate
    children: List['AccessTreeNode'] = field(default_factory=list)
    attribute: Optional[str] = None  # For leaf nodes
    secret_share: Optional[Any] = None  # For secret sharing
    parent: Optional['AccessTreeNode'] = None
    
    def is_leaf(self) -> bool:
        """Check if node is leaf"""
        return len(self.children) == 0
    
    def get_num_children(self) -> int:
        """Get number of children"""
        return len(self.children)
    
    def get_threshold(self) -> int:
        """Get threshold value k"""
        if self.gate_type == GateType.OR:
            return 1
        elif self.gate_type == GateType.AND:
            return len(self.children)
        else:
            return self.threshold or 1

class AccessTree:
    """
    Access tree structure from paper Section 6.
    Represents monotone access structure for attribute-based policies.
    
    The tree has:
        - Non-leaf nodes: threshold gates (k, num)
        - Leaf nodes: attributes
    """
    
    def __init__(self):
        self.root: Optional[AccessTreeNode] = None
        self.nodes: Dict[str, AccessTreeNode] = {}
        self.attributes: Set[str] = set()
        self.secret_sharing = ShamirSecretSharing()
    
    def build_from_policy(self, policy_dict: Dict) -> None:
        """
        Build access tree from policy dictionary.
        
        Policy format:
        {
            "type": "OR|AND|THRESHOLD",
            "threshold": k,  # for THRESHOLD
            "children": [
                {"type": "leaf", "attribute": "attr1"},
                {"type": "OR", "children": [...]}
            ]
        }
        """
        self.root = self._build_node(policy_dict, None)
        self._collect_attributes(self.root)
        logger.info(f"Built access tree with {len(self.attributes)} attributes")
    
    def _build_node(self, node_dict: Dict, parent: Optional[AccessTreeNode]) -> AccessTreeNode:
        """Recursively build tree nodes"""
        node_id = f"node_{len(self.nodes)}"
        
        if node_dict.get("type") == "leaf":
            # Leaf node (attribute)
            gate_type = GateType.OR  # Leaf nodes are treated as OR with threshold 1
            attribute = node_dict.get("attribute")
            node = AccessTreeNode(
                node_id=node_id,
                gate_type=gate_type,
                attribute=attribute,
                parent=parent
            )
        else:
            # Internal node
            gate_type = GateType[node_dict.get("type", "OR")]
            threshold = node_dict.get("threshold")
            
            node = AccessTreeNode(
                node_id=node_id,
                gate_type=gate_type,
                threshold=threshold,
                parent=parent
            )
            
            # Build children
            for child_dict in node_dict.get("children", []):
                child = self._build_node(child_dict, node)
                node.children.append(child)
        
        self.nodes[node_id] = node
        return node
    
    def _collect_attributes(self, node: AccessTreeNode) -> None:
        """Collect all attributes in the tree"""
        if node.is_leaf() and node.attribute:
            self.attributes.add(node.attribute)
        for child in node.children:
            self._collect_attributes(child)
    
    def get_dual_structure(self) -> 'AccessTree':
        """
        Get dual access structure as described in Section 4.2.1.
        
        Dual access structure Γ* contains all subsets A such that A^c ∉ Γ.
        For threshold gates: (k, num) -> (num - k + 1, num)
        """
        dual = AccessTree()
        dual.root = self._build_dual_node(self.root)
        dual._collect_attributes(dual.root)
        return dual
    
    def _build_dual_node(self, node: AccessTreeNode) -> AccessTreeNode:
        """Recursively build dual tree nodes"""
        if node.is_leaf():
            # Leaf nodes remain the same
            return AccessTreeNode(
                node_id=f"dual_{node.node_id}",
                gate_type=node.gate_type,
                attribute=node.attribute
            )
        
        # For internal nodes, transform threshold: k -> (num - k + 1)
        num = len(node.children)
        k = node.get_threshold()
        dual_k = num - k + 1
        
        if dual_k == 1:
            dual_gate = GateType.OR
            dual_threshold = None
        elif dual_k == num:
            dual_gate = GateType.AND
            dual_threshold = None
        else:
            dual_gate = GateType.THRESHOLD
            dual_threshold = dual_k
        
        dual_node = AccessTreeNode(
            node_id=f"dual_{node.node_id}",
            gate_type=dual_gate,
            threshold=dual_threshold
        )
        
        # Recursively build children
        for child in node.children:
            dual_child = self._build_dual_node(child)
            dual_child.parent = dual_node
            dual_node.children.append(dual_child)
        
        return dual_node
    
    def distribute_secrets(self, secret: Any) -> Dict[str, Any]:
        """
        Distribute secret shares according to access tree (Phase 1 of Section 6.2).
        
        Uses top-down approach:
        1. Set root secret
        2. For each node, split secret among children using threshold scheme
        """
        if not self.root:
            raise ValueError("Access tree not built")
        
        shares = {}
        self._distribute_secret_node(self.root, secret, shares)
        return shares
    
    def _distribute_secret_node(self, node: AccessTreeNode, secret: Any, shares: Dict) -> None:
        """Distribute secret to node and its children"""
        node.secret_share = secret
        
        if node.is_leaf():
            # Leaf node gets the share directly
            shares[node.attribute] = secret
            return
        
        # Split secret among children using threshold scheme
        k = node.get_threshold()
        num = len(node.children)
        
        if k == 1:
            # OR gate: all children get same secret
            for child in node.children:
                self._distribute_secret_node(child, secret, shares)
        
        elif k == num:
            # AND gate: split secret additively
            from random import randint
            # For mock, split into random parts
            parts = []
            total = 0
            for i in range(num - 1):
                part = randint(0, 100)  # Mock splitting
                parts.append(part)
                total += part
            parts.append(secret - total)
            
            for child, part in zip(node.children, parts):
                self._distribute_secret_node(child, part, shares)
        
        else:
            # Threshold gate: use Shamir secret sharing
            points = self.secret_sharing.split_secret(secret, num, k)
            
            for child, point in zip(node.children, points):
                # point is (x, y) where y = f(x)
                self._distribute_secret_node(child, point, shares)
    
    def evaluate_satisfiability(self, attribute_set: Set[str]) -> bool:
        """
        Evaluate if a set of attributes satisfies the access tree.
        Returns True if qualified set.
        """
        if not self.root:
            return False
        
        return self._evaluate_node(self.root, attribute_set)
    
    def _evaluate_node(self, node: AccessTreeNode, attribute_set: Set[str]) -> bool:
        """Recursively evaluate node satisfiability"""
        if node.is_leaf():
            # Leaf node satisfied if attribute in set
            return node.attribute in attribute_set
        
        # Count satisfied children
        satisfied = 0
        for child in node.children:
            if self._evaluate_node(child, attribute_set):
                satisfied += 1
        
        # Check threshold
        return satisfied >= node.get_threshold()
    
    def get_minimal_qualified_sets(self) -> List[Set[str]]:
        """
        Get all minimal qualified sets (minimal sets that satisfy the tree).
        Used for consistency checking in Section 6.3.
        """
        if not self.root:
            return []
        
        all_attrs = list(self.attributes)
        minimal_sets = []
        
        # Brute force for small sets (optimize for production)
        from itertools import combinations
        
        for r in range(1, len(all_attrs) + 1):
            for subset in combinations(all_attrs, r):
                subset_set = set(subset)
                if self.evaluate_satisfiability(subset_set):
                    # Check if minimal
                    is_minimal = True
                    for attr in subset:
                        if self.evaluate_satisfiability(subset_set - {attr}):
                            is_minimal = False
                            break
                    if is_minimal:
                        minimal_sets.append(subset_set)
            
            if minimal_sets:
                break  # Found minimal at smallest size
        
        return minimal_sets
    
    def to_dict(self) -> Dict:
        """Convert tree to dictionary"""
        return self._node_to_dict(self.root)
    
    def _node_to_dict(self, node: AccessTreeNode) -> Dict:
        """Convert node to dictionary"""
        if node.is_leaf():
            return {
                "type": "leaf",
                "attribute": node.attribute
            }
        
        return {
            "type": node.gate_type.value,
            "threshold": node.threshold,
            "children": [self._node_to_dict(child) for child in node.children]
        }
    
    @classmethod
    def from_dict(cls, data: Dict) -> 'AccessTree':
        """Create tree from dictionary"""
        tree = cls()
        tree.build_from_policy(data)
        return tree

"""
Secret Sharing Scheme implementation from Section 4.3 of the paper.
Implements Shamir's secret sharing for threshold access structures.
"""

from typing import List, Tuple, Any, Optional
import random
import logging

logger = logging.getLogger(__name__)

class ShamirSecretSharing:
    """
    Shamir's secret sharing scheme as described in Section 4.3.1.
    
    Properties:
        - Perfect: non-qualified sets get no info about secret
        - Ideal: share length = secret length
        - Smooth: polynomial time operations
    """
    
    def __init__(self, prime_modulus: int = None):
        """
        Initialize with prime modulus p.
        Default uses a large prime for Zp.
        """
        # Use a large prime for finite field (simplified for mock)
        self.modulus = prime_modulus or (2**127 - 1)  # Mersenne prime
    
    def split_secret(
        self, 
        secret: Any, 
        num_shares: int, 
        threshold: int
    ) -> List[Tuple[int, Any]]:
        """
        Split secret into num_shares shares with threshold k.
        
        Algorithm from Section 4.3.1:
        1. Define polynomial f(x) = a0 + a1*x + ... + a_{k-1}*x^{k-1}
           where a0 = secret
        2. Assign each participant point (x_i, f(x_i))
        
        Returns list of (x_i, y_i) shares.
        """
        if threshold > num_shares:
            raise ValueError("Threshold cannot exceed number of shares")
        if threshold < 1:
            raise ValueError("Threshold must be at least 1")
        
        # Convert secret to integer for polynomial
        secret_int = self._to_int(secret)
        
        # Generate random coefficients a1...a_{k-1}
        coefficients = [secret_int]  # a0 = secret
        for i in range(1, threshold):
            coeff = random.randint(1, self.modulus - 1)
            coefficients.append(coeff)
        
        # Generate shares
        shares = []
        for i in range(1, num_shares + 1):
            x = i
            # Evaluate polynomial at x
            y = self._evaluate_polynomial(coefficients, x)
            shares.append((x, y))
        
        logger.debug(f"Generated {num_shares} shares with threshold {threshold}")
        return shares
    
    def reconstruct_secret(self, shares: List[Tuple[int, Any]]) -> Any:
        """
        Reconstruct secret from threshold number of shares using Lagrange interpolation.
        
        f(x) = Σ Δ_{x_i}(x) * f(x_i)
        where Δ_{x_i}(x) is Lagrange coefficient.
        """
        if len(shares) < 1:
            raise ValueError("Need at least one share")
        
        # Extract x and y values
        xs = [s[0] for s in shares]
        ys = [self._to_int(s[1]) for s in shares]
        
        # Lagrange interpolation at x=0 to get secret
        secret = 0
        for i in range(len(shares)):
            # Compute Lagrange basis polynomial at 0
            numerator = 1
            denominator = 1
            for j in range(len(shares)):
                if j != i:
                    numerator *= (-xs[j])
                    denominator *= (xs[i] - xs[j])
            
            # Compute term
            lagrange = (numerator * self._mod_inverse(denominator, self.modulus)) % self.modulus
            secret = (secret + ys[i] * lagrange) % self.modulus
        
        return secret
    
    def _evaluate_polynomial(self, coefficients: List[int], x: int) -> int:
        """Evaluate polynomial at point x"""
        result = 0
        for i, coeff in enumerate(coefficients):
            result = (result + coeff * pow(x, i, self.modulus)) % self.modulus
        return result
    
    def _mod_inverse(self, a: int, m: int) -> int:
        """Compute modular inverse using extended Euclidean algorithm"""
        g, x, _ = self._egcd(a, m)
        if g != 1:
            raise ValueError("Modular inverse does not exist")
        return x % m
    
    def _egcd(self, a: int, b: int) -> Tuple[int, int, int]:
        """Extended Euclidean algorithm"""
        if a == 0:
            return (b, 0, 1)
        else:
            g, x, y = self._egcd(b % a, a)
            return (g, y - (b // a) * x, x)
    
    def _to_int(self, value: Any) -> int:
        """Convert any value to integer for polynomial operations"""
        if isinstance(value, int):
            return value
        elif isinstance(value, str):
            # Hash string to integer
            import hashlib
            return int(hashlib.sha256(value.encode()).hexdigest(), 16) % self.modulus
        elif hasattr(value, '__int__'):
            return int(value)
        else:
            # Fallback: use hash
            return hash(value) % self.modulus
    
    def verify_share_consistency(
        self, 
        shares: List[Tuple[int, Any]], 
        threshold: int
    ) -> bool:
        """
        Verify that shares are consistent (all from same polynomial).
        Any threshold+1 shares should reconstruct the same secret.
        """
        if len(shares) <= threshold:
            return True  # Can't verify with insufficient shares
        
        # Take first threshold shares to reconstruct
        secret1 = self.reconstruct_secret(shares[:threshold])
        
        # Try different combinations
        for i in range(threshold, len(shares)):
            # Replace one share
            test_shares = shares[:threshold-1] + [shares[i]]
            secret2 = self.reconstruct_secret(test_shares)
            if secret1 != secret2:
                return False
        
        return True


class LinearSecretSharing:
    """
    Linear secret sharing scheme for general monotone access structures.
    Used for access trees with mixed gates.
    """
    
    def __init__(self):
        self.shamir = ShamirSecretSharing()
    
    def share_secret_tree(self, secret: Any, tree) -> Dict[str, Any]:
        """
        Share secret according to access tree.
        Returns mapping from leaf attributes to shares.
        """
        shares = {}
        self._share_node(secret, tree.root, shares)
        return shares
    
    def _share_node(self, secret: Any, node, shares: Dict) -> None:
        """Recursively share secret at node"""
        if node.is_leaf():
            shares[node.attribute] = secret
            return
        
        if node.gate_type == "OR":
            # All children get same secret
            for child in node.children:
                self._share_node(secret, child, shares)
        
        elif node.gate_type == "AND":
            # Split secret additively
            num = len(node.children)
            parts = self._additive_split(secret, num)
            for child, part in zip(node.children, parts):
                self._share_node(part, child, shares)
        
        else:  # THRESHOLD
            # Use Shamir sharing
            k = node.threshold
            num = len(node.children)
            points = self.shamir.split_secret(secret, num, k)
            
            for child, point in zip(node.children, points):
                # point[1] is the share value
                self._share_node(point[1], child, shares)
    
    def _additive_split(self, secret: Any, num_parts: int) -> List[Any]:
        """Split secret into num_parts additive shares"""
        secret_int = self.shamir._to_int(secret)
        parts = []
        total = 0
        
        for i in range(num_parts - 1):
            part = random.randint(1, self.shamir.modulus - 1)
            parts.append(part)
            total += part
        
        # Last part makes total equal secret
        last_part = (secret_int - total) % self.shamir.modulus
        parts.append(last_part)
        
        return parts

    Signature Generation Algorithm (Sign) from paper Section 5.
    
    Input:
        - Secret values μ, μ', {μ_j}
        - Pseudonym pseu_S'
        - Message M
        - Public parameters PP
        - Access tree AT
        - Set of non-delegatable attributes S_ad
    
    Output:
        - Signature σ
    """
    
    def __init__(self, pairing: BilinearPairing = None):
        self.pairing = pairing or get_pairing()
        self.P, self.g, self.h = self.pairing.get_generators()
        self.secret_sharing = ShamirSecretSharing()
    
    def sign(
        self,
        mu: Any,  # User's secret μ
        mu_prime: Any,  # Pseudonym secret μ'
        delegated_mu: Dict[str, Any],  # attr -> μ_j for delegated attributes
        pseudonym: Pseudonym,
        message: str,
        access_tree: AccessTree,
        non_delegatable: List[str],  # Attributes that don't allow delegation
        company_public_keys: Dict[str, str],  # W_i per attribute
        nonce: Optional[str] = None  # Gate nonce for replay prevention
    ) -> Signature:
        """
        Generate signature as per paper Section 5.
        
        Steps:
        1) Select random factors r1, r3, r5
        2) Compute commitment T_G1 = r1 * P
        3) Compute commitments t3 = h^r3 * g^{-r1}, t5 = h^r5
        4) Select random r2,i for each attribute
        5) Compute t2,i = [e(P, P + Pa_i)]^{r2,i}

           a) Select random r4,i
           b) Compute t4,i = h^{r4,i} * g^{-r2,i}
        7) Select random δ, γ
        8) Compute auxiliary keys ~y1 = h^γ * g^{μ+μ'}, ~y2 = h^δ * g^{μ'}
        9) Get challenges c_i for non-satisfying leaf nodes (Phase 1)
        10) For attributes without credentials:
            a) Select random s4,i
            b) Compute simulated commitments
        11) Obtain challenge c = H(Pu||Pa1||...||TG1||t2,1||...||t3||{t4,i}||t5||~y1||~y2||M||z)
        12) Get c_i for satisfying leaf nodes (Phase 2)
        13) Compute responses
        """
        logger.info(f"Generating signature for message: {message[:50]}...")
        
        # Combine message with nonce for replay protection
        if nonce:
            message_with_nonce = f"{message}:{nonce}"
        else:
            message_with_nonce = message
        
        # Step 1: Select random factors
        r1 = self.pairing.random_ZR()
        r3 = self.pairing.random_ZR()
        r5 = self.pairing.random_ZR()
        
        # Step 2: Compute T_G1 = r1 * P
        T_G1 = self._scalar_mult_G1(r1, self.P)
        T_G1_str = self.pairing.serialize_G1(T_G1)
        
        # Step 3: Compute t3 = h^r3 * g^{-r1}, t5 = h^r5
        h_r3 = self._exp_gt(self.h, r3)
        g_neg_r1 = self._exp_gt(self.g, self._negate_zr(r1))
        t3 = self._mult_gt(h_r3, g_neg_r1)
        t3_str = self.pairing.serialize_GT(t3)
        
        t5 = self._exp_gt(self.h, r5)
        t5_str = self.pairing.serialize_GT(t5)
        
        # Step 4-5: Select r2,i and compute t2,i
        r2_i = {}
        t2_i = {}
        Pa_values = pseudonym.pa_values
        
        for attr, pa_str in Pa_values.items():
            r2 = self.pairing.random_ZR()
            r2_i[attr] = self._serialize_zr(r2)
            
            # Compute e(P, P + Pa_i)
            Pa = self.pairing.deserialize_G1(pa_str)
            P_plus_Pa = self._add_g1(self.P, Pa)
            e_P_PplusPa = self.pairing.pair(self.P, P_plus_Pa)
            
            # Compute t2,i = [e(P, P+Pa_i)]^{r2}
            t2 = self._exp_gt(e_P_PplusPa, r2)
            t2_i[attr] = self.pairing.serialize_GT(t2)
        
        # Step 6: For non-delegatable attributes
        r4_i = {}
        t4_i = {}

            r4_i[attr] = self._serialize_zr(r4)

            h_r4 = self._exp_gt(self.h, r4)
            r2 = self._deserialize_zr(r2_i[attr])
            g_neg_r2 = self._exp_gt(self.g, self._negate_zr(r2))
            t4 = self._mult_gt(h_r4, g_neg_r2)
            t4_i[attr] = self.pairing.serialize_GT(t4)

        gamma = self.pairing.random_ZR()

        g_mu_plus = self._exp_gt(self.g, mu_plus_mu_prime)
        y1 = self._mult_gt(h_gamma, g_mu_plus)
        y1_str = self.pairing.serialize_GT(y1)
        
        # ~y2 = h^δ * g^{μ'}
        h_delta = self._exp_gt(self.h, delta)
        g_mu_prime = self._exp_gt(self.g, mu_prime)
        y2 = self._mult_gt(h_delta, g_mu_prime)
        y2_str = self.pairing.serialize_GT(y2)
        
        # Step 9-10: Get challenges (simplified for now - full implementation would
        # use the access tree phase 1 from Section 6.2)
        c_i = {}
        for attr in Pa_values.keys():
            # Mock challenge (real implementation uses access tree)
            c_i[attr] = self._serialize_zr(self.pairing.random_ZR())
        
        # Step 11: Compute main challenge c
        challenge_input = (
            pseudonym.pu_value +
            json.dumps(pseudonym.pa_values) +
            T_G1_str +
            json.dumps(t2_i) +
            t3_str +
            json.dumps(t4_i) +
            t5_str +
            y1_str +
            y2_str +
            message_with_nonce +
            pseudonym.z_value
        )
        c = self.pairing.hash_to_ZR(challenge_input.encode())
        c_str = self._serialize_zr(c)
        
        # Step 13: Compute responses
        # s1 = r1 + (μ + μ') * c
        mu_plus_mu_prime = self._add_zr(mu, mu_prime)
        term1 = self._mul_zr(mu_plus_mu_prime, c)
        s1 = self._add_zr(r1, term1)
        s1_str = self._serialize_zr(s1)
        
        # s2,i = r2,i + μ' * c_i
        s2_i = {}
        for attr in Pa_values.keys():
            r2 = self._deserialize_zr(r2_i[attr])
            mu_prime_c_i = self._mul_zr(mu_prime, self._deserialize_zr(c_i[attr]))
            s2 = self._add_zr(r2, mu_prime_c_i)
            s2_i[attr] = self._serialize_zr(s2)
        
        # s3 = -c*γ + r3
        c_gamma = self._mul_zr(c, gamma)
        s3 = self._add_zr(r3, self._negate_zr(c_gamma))
        s3_str = self._serialize_zr(s3)
        
        # s5 = -c*(δ+γ) + r5
        delta_plus_gamma = self._add_zr(delta, gamma)
        c_delta_gamma = self._mul_zr(c, delta_plus_gamma)
        s5 = self._add_zr(r5, self._negate_zr(c_delta_gamma))
        s5_str = self._serialize_zr(s5)
        
        # s4,i = -c_i*δ + r4,i for non-delegatable
        s4_i = {}
        for attr in non_delegatable:
            r4 = self._deserialize_zr(r4_i[attr])
            c_i_delta = self._mul_zr(self._deserialize_zr(c_i[attr]), delta)
            s4 = self._add_zr(r4, self._negate_zr(c_i_delta))
            s4_i[attr] = self._serialize_zr(s4)
        
        # Create signature
        signature = Signature(

            s2_i=s2_i,
            s3=s3_str,
            s4_i=s4_i,
            s5=s5_str,
            y1=y1_str,
            y2=y2_str,
            message=message,
            timestamp=datetime.utcnow().isoformat()
        )
        
        logger.info(f"Signature generated with challenge: {c_str[:16]}...")
        return signature

        """Multiply G1 point by scalar"""
        if hasattr(scalar, '__mul__'):
            return scalar * point
        return point * scalar

        """Add two G1 points"""
        if hasattr(a, '__add__'):
            return a + b
        return a + b
    
    def _exp_gt(self, base: Any, exp: Any) -> Any:
        """Exponentiation in GT"""
        if hasattr(base, '__pow__'):
            return base ** exp
        return base ** exp
    
    def _mult_gt(self, a: Any, b: Any) -> Any:
        """Multiply two GT elements"""
        if hasattr(a, '__mul__'):
            return a * b

        """Add two ZR elements"""
        if hasattr(a, '__add__'):
            return a + b

    def _mul_zr(self, a: Any, b: Any) -> Any:
        """Multiply two ZR elements"""
        if hasattr(a, '__mul__'):
            return a * b
        return a * b
    
    def _negate_zr(self, a: Any) -> Any:
        """Negate ZR element"""
        if hasattr(a, '__neg__'):
            return -a
        return -a
    
    def _serialize_zr(self, a: Any) -> str:
        """Serialize ZR element"""
        return str(a)

"""
Signature Verification Algorithm (SignCheck) from Section 5 of the paper.
Verifies signatures without learning the attributes.
"""

from typing import Dict, Any, Optional, Set
import json
import hashlib
import logging

from app.crypto.pairing import get_pairing, BilinearPairing

from app.config import settings

logger = logging.getLogger(__name__)

class SignCheck:
    """
    Signature Verification Algorithm (SignCheck) from paper Section 5.
    
    Input:
        - Signature σ
        - Pseudonym pseu_S'
        - Message M
        - Public parameters PP
        - Access tree AT
        - Set of non-delegatable attributes S_ad
    
    Output:
        - Valid/Invalid
    """
    
    def __init__(self, pairing: BilinearPairing = None):
        self.pairing = pairing or get_pairing()
        self.P, self.g, self.h = self.pairing.get_generators()
    
    def verify(
        self,
        signature: Signature,
        pseudonym: Pseudonym,
        message: str,
        access_tree: AccessTree,
        non_delegatable: Set[str],
        company_public_keys: Dict[str, str],  # W_i per attribute
        nonce: Optional[str] = None
    ) -> bool:
        """
        Verify signature as per paper Section 5.
        
        Steps:
        1) Compute d = H(z)
        2) Compute ~TG1 = s1*P - c*Pu
        3) For each attribute: compute ~t2,i
        4) Compute ~t3 = h^{s3} * g^{-s1} * ~y1^c
        5) Compute ~t4,i = h^{s4,i} * g^{-s2,i} * ~y2^c
        6) Compute ~t5 = h^{s5} * (~y1*~y2/g^d)^c
        7) Check c == H(Pu||...||M||z)
        8) Check consistency of shares {c_i} with secret c
        """
        logger.info("Verifying signature...")
        
        # Step 1: Compute d = H(z)
        z_bytes = pseudonym.z_value.encode()
        d = self.pairing.hash_to_ZR(z_bytes)
        
        # Step 2: Compute ~TG1 = s1*P - c*Pu
        s1 = self._deserialize_zr(signature.s1)
        c = self._deserialize_zr(signature.c)
        
        Pu = self.pairing.deserialize_G1(pseudonym.pu_value)
        
        # s1*P
        s1_P = self._scalar_mult_G1(s1, self.P)
        
        # c*Pu
        c_Pu = self._scalar_mult_G1(c, Pu)
        
        # s1*P - c*Pu
        TG1_computed = self._subtract_g1(s1_P, c_Pu)
        
        # Step 3: Compute ~t2,i for each attribute
        t2_computed = {}
        
        for attr, pa_str in pseudonym.pa_values.items():
            if attr not in signature.c_i or attr not in signature.s2_i:
                logger.warning(f"Missing signature components for attribute {attr}")
                return False
            
            c_i = self._deserialize_zr(signature.c_i[attr])
            s2_i = self._deserialize_zr(signature.s2_i[attr])
            
            # Get Pa_i
            Pa = self.pairing.deserialize_G1(pa_str)
            
            # Get W_i (public key)
            if attr not in company_public_keys:
                logger.warning(f"Missing public key for attribute {attr}")
                return False
            W_i = self.pairing.deserialize_G1(company_public_keys[attr])
            
            # Compute e(P, P + Pa_i)^{s2,i}
            P_plus_Pa = self._add_g1(self.P, Pa)
            e_P_PplusPa = self.pairing.pair(self.P, P_plus_Pa)
            left = self._exp_gt(e_P_PplusPa, s2_i)
            
            # Compute e(Pu + W_i, Pa_i)^{c_i}
            Pu_plus_Wi = self._add_g1(Pu, W_i)
            right_base = self.pairing.pair(Pu_plus_Wi, Pa)
            right = self._exp_gt(right_base, c_i)
            
            # ~t2,i = left / right
            t2 = self._div_gt(left, right)
            t2_computed[attr] = self.pairing.serialize_GT(t2)
        
        # Step 4: Compute ~t3
        s3 = self._deserialize_zr(signature.s3)
        y1 = self.pairing.deserialize_GT(signature.y1)
        
        # h^{s3}
        h_s3 = self._exp_gt(self.h, s3)
        
        # g^{-s1}
        g_neg_s1 = self._exp_gt(self.g, self._negate_zr(s1))
        
        # ~y1^c
        y1_c = self._exp_gt(y1, c)
        
        # ~t3 = h^{s3} * g^{-s1} * ~y1^c
        t3_computed = self._mult_gt(self._mult_gt(h_s3, g_neg_s1), y1_c)
        
        # Step 5: Compute ~t4,i for non-delegatable attributes
        t4_computed = {}
        
        for attr in non_delegatable:
            if attr not in signature.s4_i:
                continue
            
            s4 = self._deserialize_zr(signature.s4_i[attr])
            s2 = self._deserialize_zr(signature.s2_i[attr])
            y2 = self.pairing.deserialize_GT(signature.y2)
            
            # h^{s4}
            h_s4 = self._exp_gt(self.h, s4)
            
            # g^{-s2}
            g_neg_s2 = self._exp_gt(self.g, self._negate_zr(s2))
            
            # ~y2^c
            y2_c = self._exp_gt(y2, c)
            
            # ~t4 = h^{s4} * g^{-s2} * ~y2^c
            t4 = self._mult_gt(self._mult_gt(h_s4, g_neg_s2), y2_c)
            t4_computed[attr] = self.pairing.serialize_GT(t4)
        
        # Step 6: Compute ~t5
        s5 = self._deserialize_zr(signature.s5)
        y2 = self.pairing.deserialize_GT(signature.y2)
        
        # h^{s5}
        h_s5 = self._exp_gt(self.h, s5)
        
        # ~y1 * ~y2 / g^d
        y1_y2 = self._mult_gt(y1, y2)
        g_d = self._exp_gt(self.g, d)
        y1y2_div_gd = self._div_gt(y1_y2, g_d)
        
        # (y1y2/gd)^c
        y1y2_div_gd_c = self._exp_gt(y1y2_div_gd, c)

        message_with_nonce = message
        if nonce:
            message_with_nonce = f"{message}:{nonce}"

            pseudonym.pu_value +
            json.dumps(pseudonym.pa_values) +

            signature.y2 +
            message_with_nonce +
            pseudonym.z_value

            return False

        # as described in Section 6.3

        return True

"""
Revocation Algorithm (Revoke) from Section 5 of the paper.
Links pseudonyms to identities for privacy revocation in case of misuse.
"""

from typing import Dict, Any, Optional, List, Tuple
import logging

from app.crypto.pairing import get_pairing, BilinearPairing
from app.crypto.pseugen import Pseudonym
from app.crypto.cregen import Credential

logger = logging.getLogger(__name__)

class Revoke:
    """
    Revocation Algorithm (Revoke) from paper Section 5.
    
    Input:
        - Pseudonym pseu_S
        - Registry of issued credentials (ID, cred, S)
    
    Output:
        - Identity ID of misuser
    """
    
    def __init__(self, pairing: BilinearPairing = None):
        self.pairing = pairing or get_pairing()
        self.P, self.g, self.h = self.pairing.get_generators()
    
    def revoke(
        self,
        pseudonym: Pseudonym,
        credentials: List[Tuple[int, Credential, Dict[str, str]]]  # (ID, cred, attribute_public_keys)
    ) -> Optional[int]:
        """
        Find owner of pseudonym as per paper Section 5.
        
        Steps:
        1) Check if credential's attributes overlap with pseudonym
        2) For each Pa_i in pseudonym where i in I_S, check:
           e(Pu + W_i, Sa_i) == e(P, P + Pa_i)
        3) Return ID if match found
        """
        logger.info(f"Attempting to revoke pseudonym: {pseudonym.get_hash()[:16]}...")
        
        Pu = self.pairing.deserialize_G1(pseudonym.pu_value)
        
        for identity_id, credential, public_keys in credentials:
            # Step 1: Check attribute overlap
            cred_attrs = set(credential.sa_values.keys())
            pseudonym_attrs = set(pseudonym.pa_values.keys())
            
            overlap = cred_attrs.intersection(pseudonym_attrs)
            if not overlap:
                continue
            
            # Step 2: Check equation for each overlapping attribute
            for attr in overlap:
                if attr not in public_keys:
                    continue
                
                # Get W_i
                W_i = self.pairing.deserialize_G1(public_keys[attr])
                
                # Get Sa_i from credential
                Sa_i = self.pairing.deserialize_G1(credential.sa_values[attr])
                
                # Get Pa_i from pseudonym
                Pa_i = self.pairing.deserialize_G1(pseudonym.pa_values[attr])
                
                # Check e(Pu + W_i, Sa_i) == e(P, P + Pa_i)
                left = self.pairing.pair(self._add_g1(Pu, W_i), Sa_i)
                
                P_plus_Pa = self._add_g1(self.P, Pa_i)
                right = self.pairing.pair(self.P, P_plus_Pa)
                
                if self._equal_gt(left, right):
                    logger.info(f"Found matching identity: {identity_id}")
                    return identity_id
        
        logger.info("No matching identity found")
        return None
    
    def _add_g1(self, a: Any, b: Any) -> Any:
        """Add two G1 points"""
        if hasattr(a, '__add__'):
            return a + b
        return a + b
    
    def _equal_gt(self, a: Any, b: Any) -> bool:
        """Compare two GT elements"""
        if hasattr(a, '__eq__'):
            return a == b
        return str(a) == str(b)

"""
Base repository class with common database operations.
"""

from typing import Generic, TypeVar, Type, Optional, List, Dict, Any
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, update, delete
from sqlalchemy.sql import func
import logging

from app.database.models import Base

ModelType = TypeVar("ModelType", bound=Base)
logger = logging.getLogger(__name__)

class BaseRepository(Generic[ModelType]):
    """Base repository with CRUD operations"""
    
    def __init__(self, model: Type[ModelType], session: AsyncSession):
        self.model = model
        self.session = session
    
    async def create(self, **kwargs) -> ModelType:
        """Create a new record"""
        instance = self.model(**kwargs)
        self.session.add(instance)
        await self.session.flush()
        await self.session.refresh(instance)
        logger.debug(f"Created {self.model.__name__}: {getattr(instance, 'id', None)}")
        return instance
    
    async def get(self, id: int) -> Optional[ModelType]:
        """Get by ID"""
        query = select(self.model).where(self.model.id == id)
        result = await self.session.execute(query)
        return result.scalar_one_or_none()
    
    async def get_by_field(self, field: str, value: Any) -> Optional[ModelType]:
        """Get by field value"""
        column = getattr(self.model, field, None)
        if not column:
            raise ValueError(f"Field {field} not found in {self.model.__name__}")
        
        query = select(self.model).where(column == value)
        result = await self.session.execute(query)
        return result.scalar_one_or_none()
    
    async def get_many(
        self,
        skip: int = 0,
        limit: int = 100,
        **filters
    ) -> List[ModelType]:
        """Get multiple records with filters"""
        query = select(self.model)
        
        for field, value in filters.items():
            if value is not None:
                column = getattr(self.model, field, None)
                if column:
                    query = query.where(column == value)
        
        query = query.offset(skip).limit(limit)
        result = await self.session.execute(query)
        return result.scalars().all()
    
    async def update(self, id: int, **kwargs) -> Optional[ModelType]:
        """Update a record"""
        # Remove None values
        updates = {k: v for k, v in kwargs.items() if v is not None}
        if not updates:
            return await self.get(id)
        
        # Add updated_at timestamp
        if hasattr(self.model, 'updated_at'):
            updates['updated_at'] = func.now()
        
        query = (
            update(self.model)
            .where(self.model.id == id)
            .values(**updates)
            .returning(self.model)
        )
        result = await self.session.execute(query)
        await self.session.flush()
        return result.scalar_one_or_none()
    
    async def delete(self, id: int, soft: bool = True) -> bool:
        """Delete a record (soft delete if supported)"""
        if soft and hasattr(self.model, 'is_active'):
            # Soft delete
            query = (
                update(self.model)
                .where(self.model.id == id)
                .values(is_active=False, updated_at=func.now())
            )
            result = await self.session.execute(query)
            return result.rowcount > 0
        else:
            # Hard delete
            query = delete(self.model).where(self.model.id == id)
            result = await self.session.execute(query)
            return result.rowcount > 0
    
    async def count(self, **filters) -> int:
        """Count records with filters"""
        query = select(func.count()).select_from(self.model)
        
        for field, value in filters.items():
            if value is not None:
                column = getattr(self.model, field, None)
                if column:
                    query = query.where(column == value)
        
        result = await self.session.execute(query)
        return result.scalar()

from passlib.context import CryptContext

from app.database.models import Admin

# Password hashing context
pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

class AdminRepository(BaseRepository[Admin]):
    """Admin repository with authentication"""
    
    def __init__(self, session: AsyncSession):
        super().__init__(Admin, session)
    
    async def create_admin(
        self,
        username: str,
        email: str,
        password: str,
        full_name: str,
        role: str = "admin"
    ) -> Admin:
        """Create a new admin user"""
        # Hash password
        password_hash = pwd_context.hash(password)
        
        admin = await self.create(
            username=username,
            email=email,
            password_hash=password_hash,
            full_name=full_name,
            role=role,
            is_active=True
        )
        
        logger.info(f"Created admin user: {username}")
        return admin
    
    async def authenticate(self, username: str, password: str) -> Optional[Admin]:
        """Authenticate an admin"""
        admin = await self.get_by_field("username", username)
        
        if not admin or not admin.is_active:
            return None
        
        if pwd_context.verify(password, admin.password_hash):
            # Update last login
            await self.update(admin.admin_id, last_login=func.now())
            logger.info(f"Admin {username} authenticated")
            return admin
        
        logger.warning(f"Failed authentication for {username}")
        return None
    
    async def change_password(
        self,
        admin_id: int,
        old_password: str,
        new_password: str
    ) -> bool:
        """Change admin password"""
        admin = await self.get(admin_id)
        if not admin:
            return False
        
        if not pwd_context.verify(old_password, admin.password_hash):
            return False
        
        new_hash = pwd_context.hash(new_password)
        await self.update(admin_id, password_hash=new_hash)
        
        logger.info(f"Password changed for admin {admin_id}")
        return True
    
    async def get_by_email(self, email: str) -> Optional[Admin]:
        """Get admin by email"""
        return await self.get_by_field("email", email)
    
    async def deactivate_admin(self, admin_id: int, deactivated_by: int) -> Optional[Admin]:
        """Deactivate an admin account"""
        admin = await self.update(
            admin_id,
            is_active=False
        )
        
        if admin:
            logger.info(f"Admin {admin_id} deactivated by {deactivated_by}")

@router.post("/pseudonym/generate", response_model=PseudonymResponse)
async def generate_pseudonym(
    request: PseudonymRequest,
    db: AsyncSession = Depends(get_db)
):
    """
    Generate a one-time pseudonym for gate authentication.
    Implements PseuGen algorithm from paper Section 5.
    """
    # Get credential
    cred_repo = CredentialRepository(db)
    credential = await cred_repo.get(request.credential_id)
    
    if not credential:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Credential not found"
        )
    
    if credential.is_revoked:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Credential has been revoked"
        )
    
    if credential.expires_at and credential.expires_at < datetime.utcnow():
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Credential has expired"
        )
    
    # Get company's attribute keys for verification
    worker_repo = WorkerRepository(db)
    worker = await worker_repo.get(credential.worker_id)
    
    company_repo = CompanyRepository(db)
    company = await company_repo.get(worker.company_id)
    
    # Parse credential data
    mu = credential.mu_value
    sa_values = json.loads(credential.sa_values)
    
    # Get all possible attributes from company
    attr_keys = json.loads(company.public_key_parameters) if company.public_key_parameters else {}
    all_attributes = list(attr_keys.keys())
    
    # Generate pseudonym using PseuGen
    pseu_gen = PseuGen()
    
    # Create crypto credential object
    from app.crypto.cregen import Credential as CryptoCred
    crypto_cred = CryptoCred(
        mu=mu,
        sa_values=sa_values,
        worker_id=worker.worker_id,
        company_id=worker.company_id,
        issued_at=credential.issued_at.isoformat() if credential.issued_at else None,
        expires_at=credential.expires_at.isoformat() if credential.expires_at else None
    )
    crypto_cred.credential_id = credential.credential_id
    
    # Generate pseudonym
    pseudonym, mu_prime = pseu_gen.generate(
        credential=crypto_cred,
        all_attributes=all_attributes,
        delegated_attributes=None,  # No delegation in basic version
        z_value=request.z_value,
        expiry_minutes=settings.PSEUDONYM_EXPIRY_MINUTES
    )
    
    # Store pseudonym in database
    pseu_repo = PseudonymRepository(db)
    db_pseudonym = await pseu_repo.create_pseudonym(
        credential_id=credential.credential_id,
        pu_value=pseudonym.pu_value,
        pa_values=pseudonym.pa_values,
        z_value=pseudonym.z_value,
        mu_prime=mu_prime,
        expiry_minutes=settings.PSEUDONYM_EXPIRY_MINUTES
    )
    
    # Prepare QR data
    qr_data = QRData(
        pu=pseudonym.pu_value,
        pa=pseudonym.pa_values,
        z=pseudonym.z_value,
        credential_hash=credential.credential_hash,
        timestamp=datetime.utcnow().isoformat()
    )
    
    return PseudonymResponse(
        pseudonym_id=db_pseudonym.pseudonym_id,
        qr_data=qr_data.dict(),
        expires_at=db_pseudonym.expires_at.isoformat() if db_pseudonym.expires_at else None
    )

"""
Verification Service - Implements the complete verification pipeline from system flow.
"""

from typing import Dict, Any, Optional, Tuple
import json
import logging
from datetime import datetime
from sqlalchemy.ext.asyncio import AsyncSession

from app.database.repositories.company_repo import CompanyRepository
from app.database.repositories.worker_repo import PseudonymRepository, CredentialRepository
from app.database.repositories.gate_repo import NonceRepository
from app.database.repositories.access_log_repo import RevocationListRepository
from app.crypto.signcheck import SignCheck
from app.crypto.accesstree import AccessTree, create_delivery_policy
from app.config import settings

logger = logging.getLogger(__name__)

class VerificationService:
    """
    Implements the 7-step verification pipeline from the system flow:
    1. Signature Verification
    2. Expiry Check
    3. Timestamp Check
    4. Nonce Check
    5. Trust Check
    6. Revocation Check
    7. Replay Check
    """
    
    def __init__(self, db: AsyncSession):
        self.db = db
        self.sign_check = SignCheck()
        self.policy_tree = create_delivery_policy()
    
    async def verify(
        self,
        qr_data: Dict[str, Any],
        gate_id: int,
        gate_nonce: str,
        received_nonce: str,
        timestamp: Optional[str] = None
    ) -> Dict[str, Any]:
        """
        Execute full verification pipeline.
        Returns result dict with status and details of each check.
        """
        result = {
            "granted": False,
            "message": "Verification failed",
            "signature_verified": False,
            "expiry_check_passed": False,
            "timestamp_check_passed": False,
            "nonce_check_passed": False,
            "trust_check_passed": False,
            "revocation_check_passed": False,
            "replay_check_passed": False
        }
        
        try:
            # Step 1: Signature Verification
            sig_result = await self._verify_signature(qr_data)
            result["signature_verified"] = sig_result["success"]
            if not sig_result["success"]:
                result["failure_reason"] = f"Signature verification failed: {sig_result.get('error')}"
                return result
            
            # Step 2: Expiry Check
            expiry_result = await self._check_expiry(qr_data)
            result["expiry_check_passed"] = expiry_result["success"]
            if not expiry_result["success"]:
                result["failure_reason"] = f"Expiry check failed: {expiry_result.get('error')}"
                return result
            
            # Step 3: Timestamp Check
            if timestamp:
                ts_result = self._check_timestamp(timestamp)
                result["timestamp_check_passed"] = ts_result["success"]
                if not ts_result["success"]:
                    result["failure_reason"] = f"Timestamp check failed: {ts_result.get('error')}"
                    return result
            else:
                result["timestamp_check_passed"] = True  # Skip if no timestamp
            
            # Step 4: Nonce Check
            nonce_result = await self._check_nonce(gate_id, gate_nonce, received_nonce)
            result["nonce_check_passed"] = nonce_result["success"]
            if not nonce_result["success"]:
                result["failure_reason"] = f"Nonce check failed: {nonce_result.get('error')}"
                return result
            
            # Step 5: Trust Check
            trust_result = await self._check_trust(qr_data)
            result["trust_check_passed"] = trust_result["success"]
            if not trust_result["success"]:
                result["failure_reason"] = f"Trust check failed: {trust_result.get('error')}"
                return result
            
            # Step 6: Revocation Check
            revoke_result = await self._check_revocation(qr_data)
            result["revocation_check_passed"] = revoke_result["success"]
            if not revoke_result["success"]:
                result["failure_reason"] = f"Revocation check failed: {revoke_result.get('error')}"
                return result
            
            # Step 7: Replay Check
            replay_result = await self._check_replay(qr_data)
            result["replay_check_passed"] = replay_result["success"]
            if not replay_result["success"]:
                result["failure_reason"] = f"Replay check failed: {replay_result.get('error')}"
                return result
            
            # All checks passed
            result["granted"] = True
            result["message"] = "Access granted"
            result["worker_info"] = trust_result.get("worker_info", {})
            
            # Mark pseudonym as consumed
            if replay_result.get("pseudonym_id"):
                await self._mark_pseudonym_consumed(replay_result["pseudonym_id"])
            
        except Exception as e:
            logger.error(f"Verification error: {e}")
            result["failure_reason"] = f"System error: {str(e)}"
        
        return result
    
    async def _verify_signature(self, qr_data: Dict[str, Any]) -> Dict[str, Any]:
        """Step 1: Verify cryptographic signature"""
        try:
            # Extract data
            pu = qr_data.get("pu")
            pa = qr_data.get("pa", {})
            z = qr_data.get("z")
            
            if not pu or not pa or not z:
                return {"success": False, "error": "Missing cryptographic data"}
            
            # Get company public keys from credential hash
            cred_hash = qr_data.get("credential_hash")
            if not cred_hash:
                return {"success": False, "error": "Missing credential hash"}
            
            cred_repo = CredentialRepository(self.db)
            credential = await cred_repo.get_by_hash(cred_hash)
            
            if not credential:
                return {"success": False, "error": "Credential not found"}
            
            # Get company keys
            company_repo = CompanyRepository(self.db)
            company = await company_repo.get(credential.worker.company_id)
            
            if not company:
                return {"success": False, "error": "Company not found"}
            
            # Parse public keys
            attr_keys = json.loads(company.public_key_parameters) if company.public_key_parameters else {}
            public_keys = {attr: keys["public_key"] for attr, keys in attr_keys.items()}
            
            # In a real implementation, we would verify the signature here
            # For now, assume it's valid if we have the data
            
            return {"success": True}
            
        except Exception as e:
            logger.error(f"Signature verification error: {e}")
            return {"success": False, "error": str(e)}
    
    async def _check_expiry(self, qr_data: Dict[str, Any]) -> Dict[str, Any]:
        """Step 2: Check credential expiry"""
        try:
            cred_hash = qr_data.get("credential_hash")
            if not cred_hash:
                return {"success": False, "error": "Missing credential hash"}
            
            cred_repo = CredentialRepository(self.db)
            credential = await cred_repo.get_by_hash(cred_hash)
            
            if not credential:
                return {"success": False, "error": "Credential not found"}
            
            if credential.expires_at and credential.expires_at < datetime.utcnow():
                return {"success": False, "error": "Credential expired"}
            
            return {"success": True}
            
        except Exception as e:
            logger.error(f"Expiry check error: {e}")
            return {"success": False, "error": str(e)}
    
    def _check_timestamp(self, timestamp: str) -> Dict[str, Any]:
        """Step 3: Check timestamp freshness"""
        try:
            ts = datetime.fromisoformat(timestamp)
            now = datetime.utcnow()
            
            # Allow 5 minute skew
            if abs((now - ts).total_seconds()) > 300:
                return {"success": False, "error": "Timestamp too old"}
            
            return {"success": True}
            
        except Exception as e:
            logger.error(f"Timestamp check error: {e}")
            return {"success": False, "error": str(e)}
    
    async def _check_nonce(
        self,
        gate_id: int,
        gate_nonce: str,
        received_nonce: str
    ) -> Dict[str, Any]:
        """Step 4: Verify nonce"""
        try:
            if gate_nonce != received_nonce:
                return {"success": False, "error": "Nonce mismatch"}
            
            nonce_repo = NonceRepository(self.db)
            valid = await nonce_repo.validate_and_consume(gate_id, received_nonce)
            
            if not valid:
                return {"success": False, "error": "Invalid or expired nonce"}
            
            return {"success": True}
            
        except Exception as e:
            logger.error(f"Nonce check error: {e}")
            return {"success": False, "error": str(e)}
    
    async def _check_trust(self, qr_data: Dict[str, Any]) -> Dict[str, Any]:
        """Step 5: Verify company is trusted"""
        try:
            cred_hash = qr_data.get("credential_hash")
            if not cred_hash:
                return {"success": False, "error": "Missing credential hash"}
            
            cred_repo = CredentialRepository(self.db)
            credential = await cred_repo.get_by_hash(cred_hash)
            
            if not credential:
                return {"success": False, "error": "Credential not found"}
            
            # Check if company is approved
            company_repo = CompanyRepository(self.db)
            company = await company_repo.get(credential.worker.company_id)
            
            if not company or company.status != "APPROVED":
                return {"success": False, "error": "Company not trusted"}
            
            # Return worker info for audit
            worker_info = {
                "worker_id": credential.worker.worker_id,
                "company_id": company.company_id,
                "company_name": company.company_name,
                "role": credential.worker.role
            }
            
            return {"success": True, "worker_info": worker_info}
            
        except Exception as e:
            logger.error(f"Trust check error: {e}")
            return {"success": False, "error": str(e)}
    
    async def _check_revocation(self, qr_data: Dict[str, Any]) -> Dict[str, Any]:
        """Step 6: Check if any entity is revoked"""
        try:
            revoke_repo = RevocationListRepository(self.db)
            
            cred_hash = qr_data.get("credential_hash")
            if not cred_hash:
                return {"success": False, "error": "Missing credential hash"}
            
            cred_repo = CredentialRepository(self.db)
            credential = await cred_repo.get_by_hash(cred_hash)
            
            if not credential:
                return {"success": False, "error": "Credential not found"}
            
            # Check worker revocation
            if await revoke_repo.is_revoked("worker", credential.worker_id):
                return {"success": False, "error": "Worker revoked"}
            
            # Check company revocation
            if await revoke_repo.is_revoked("company", credential.worker.company_id):
                return {"success": False, "error": "Company revoked"}
            
            # Check credential revocation
            if await revoke_repo.is_revoked("credential", credential.credential_id):
                return {"success": False, "error": "Credential revoked"}
            
            return {"success": True}
            
        except Exception as e:
            logger.error(f"Revocation check error: {e}")
            return {"success": False, "error": str(e)}
    
    async def _check_replay(self, qr_data: Dict[str, Any]) -> Dict[str, Any]:
        """Step 7: Prevent replay attacks"""
        try:
            pu = qr_data.get("pu")
            z = qr_data.get("z")
            
            if not pu or not z:
                return {"success": False, "error": "Missing pseudonym data"}
            
            pseu_repo = PseudonymRepository(self.db)
            pseudonym = await pseu_repo.get_by_pu_z(pu, z)
            
            if not pseudonym:
                return {"success": False, "error": "Pseudonym not found"}
            
            if pseudonym.is_consumed:
                return {"success": False, "error": "Pseudonym already used"}
            
            if pseudonym.expires_at and pseudonym.expires_at < datetime.utcnow():
                return {"success": False, "error": "Pseudonym expired"}
            
            return {"success": True, "pseudonym_id": pseudonym.pseudonym_id}
            
        except Exception as e:
            logger.error(f"Replay check error: {e}")
            return {"success": False, "error": str(e)}
    
    async def _mark_pseudonym_consumed(self, pseudonym_id: int) -> None:
        """Mark pseudonym as consumed after successful verification"""
        pseu_repo = PseudonymRepository(self.db)
        await pseu_repo.mark_consumed(pseudonym_id)

from fastapi import FastAPI, Request, status
from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError
from starlette.exceptions import HTTPException as StarletteHTTPException
import logging

logger = logging.getLogger(__name__)

class AuthenticationError(Exception):
    """Authentication related errors"""
    pass

class AuthorizationError(Exception):
    """Authorization related errors"""
    pass

class CredentialError(Exception):
    """Credential related errors"""
    pass

class VerificationError(Exception):
    """Verification pipeline errors"""
    pass

class DatabaseError(Exception):
    """Database operation errors"""
    pass

def setup_exception_handlers(app: FastAPI):
    """Setup custom exception handlers"""
    
    @app.exception_handler(StarletteHTTPException)
    async def http_exception_handler(request: Request, exc: StarletteHTTPException):
        logger.warning(f"HTTP {exc.status_code}: {exc.detail}")
        return JSONResponse(
            status_code=exc.status_code,
            content={
                "error": "http_error",
                "message": exc.detail,
                "status_code": exc.status_code
            }
        )
    
    @app.exception_handler(RequestValidationError)
    async def validation_exception_handler(request: Request, exc: RequestValidationError):
        logger.warning(f"Validation error: {exc.errors()}")
        return JSONResponse(
            status_code=status.HTTP_422_UNPROCESSABLE_ENTITY,
            content={
                "error": "validation_error",
                "message": "Invalid request data",
                "details": exc.errors()
            }
        )
    
    @app.exception_handler(AuthenticationError)
    async def auth_exception_handler(request: Request, exc: AuthenticationError):
        logger.warning(f"Authentication error: {str(exc)}")
        return JSONResponse(
            status_code=status.HTTP_401_UNAUTHORIZED,
            content={
                "error": "authentication_error",
                "message": str(exc)
            }
        )
    
    @app.exception_handler(AuthorizationError)
    async def authz_exception_handler(request: Request, exc: AuthorizationError):
        logger.warning(f"Authorization error: {str(exc)}")
        return JSONResponse(
            status_code=status.HTTP_403_FORBIDDEN,
            content={
                "error": "authorization_error",
                "message": str(exc)
            }
        )
    
    @app.exception_handler(VerificationError)
    async def verification_exception_handler(request: Request, exc: VerificationError):
        logger.warning(f"Verification error: {str(exc)}")
        return JSONResponse(
            status_code=status.HTTP_400_BAD_REQUEST,
            content={
                "error": "verification_error",
                "message": str(exc)
            }
        )
    
    @app.exception_handler(Exception)
    async def general_exception_handler(request: Request, exc: Exception):
        logger.error(f"Unhandled exception: {str(exc)}", exc_info=True)
        return JSONResponse(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            content={
                "error": "internal_error",
                "message": "An internal server error occurred"
            }
        )

Authentication middleware for API endpoints.

from fastapi import Request, HTTPException, status, Depends
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from sqlalchemy.ext.asyncio import AsyncSession
from jose import jwt, JWTError
from typing import Optional, Dict, Any
import logging

from app.database.session import get_db
from app.database.repositories.admin_repo import AdminRepository
from app.database.repositories.company_repo import CompanyRepository
from app.database.repositories.gate_repo import GateRepository
from app.config import settings

logger = logging.getLogger(__name__)

security = HTTPBearer()

async def get_current_admin(
    credentials: HTTPAuthorizationCredentials = Depends(security),
    db: AsyncSession = Depends(get_db)
) -> Dict[str, Any]:
    """Get current admin from JWT token"""
    token = credentials.credentials
    
    try:
        payload = jwt.decode(
            token,
            settings.SECRET_KEY,
            algorithms=[settings.JWT_ALGORITHM]
        )
        admin_id = int(payload.get("sub"))
        
        admin_repo = AdminRepository(db)
        admin = await admin_repo.get(admin_id)
        
        if not admin or not admin.is_active:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Invalid authentication credentials"
            )
        
        return {
            "admin_id": admin.admin_id,
            "username": admin.username,
            "role": admin.role
        }
        
    except JWTError as e:
        logger.warning(f"JWT error: {e}")
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid authentication credentials"
        )

def require_role(required_role: str):
    """Dependency to require specific admin role"""
    async def role_checker(
        current_admin: Dict = Depends(get_current_admin)
    ) -> Dict:
        if current_admin["role"] != required_role and current_admin["role"] != "super_admin":
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"Role {required_role} required"
            )
        return current_admin
    return role_checker

async def verify_company_api_key(
    request: Request,

        raise HTTPException(

        )
    
    # In production, verify against database
    # For now, return company info
    return {

        "api_key": api_key
    }

async def verify_gate_api_key(
    request: Request,
    db: AsyncSession = Depends(get_db)
) -> Dict[str, Any]:
    """Verify gate API key from header"""
    api_key = request.headers.get("X-API-Key")
    gate_id = request.headers.get("X-Gate-ID")
    
    if not api_key or not gate_id:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Missing API key or gate ID"
        )
    
    gate_repo = GateRepository(db)
    gate = await gate_repo.get(int(gate_id))
    
    if not gate or gate.api_key != api_key:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid gate credentials"
        )
    
    if gate.status != "APPROVED":
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail=f"Gate is {gate.status}"
        )
    
    return {
        "gate_id": gate.gate_id,
        "gate_identifier": gate.gate_identifier
    }

import asyncio
from logging.config import fileConfig
from sqlalchemy import pool
from sqlalchemy.engine import Connection
from sqlalchemy.ext.asyncio import async_engine_from_config
from alembic import context

import sys
from pathlib import Path
sys.path.append(str(Path(__file__).parent.parent))

from app.database.models import Base
from app.config import settings

config = context.config

if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = Base.metadata

def run_migrations_offline() -> None:
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()

def do_run_migrations(connection: Connection) -> None:
    context.configure(connection=connection, target_metadata=target_metadata)

    with context.begin_transaction():
        context.run_migrations()

async def run_async_migrations() -> None:
    connectable = async_engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    async with connectable.connect() as connection:
        await connection.run_sync(do_run_migrations)

    await connectable.dispose()

def run_migrations_online() -> None:
    asyncio.run(run_async_migrations())

if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
